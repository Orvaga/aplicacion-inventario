### **2. Código básico en `app.py`**
```python
from flask import Flask, render_template, request, redirect, url_for, g
import sqlite3
import os

app = Flask(__name__)

# Configuración de la base de datos
DATABASE = 'database.db'

# Función para obtener la conexión a la base de datos
def get_db():
    db = getattr(g, '_database', None)
    if db is None:
        db = g._database = sqlite3.connect(DATABASE)
    return db

# Cerrar la conexión al finalizar la petición
@app.teardown_appcontext
def close_connection(exception):
    db = getattr(g, '_database', None)
    if db is not None:
        db.close()

# Crear tablas al inicio (si no existen)
def init_db():
    with app.app_context():
        db = get_db()
        cursor = db.cursor()
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS usuarios (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                nombre TEXT NOT NULL,
                email TEXT NOT NULL UNIQUE
            )
        ''')
        db.commit()

# Ruta principal: muestra usuarios y formulario para agregar nuevos
@app.route('/', methods=['GET', 'POST'])
def index():
    if request.method == 'POST':
        nombre = request.form['nombre']
        email = request.form['email']
        db = get_db()
        cursor = db.cursor()
        cursor.execute('INSERT INTO usuarios (nombre, email) VALUES (?, ?)', (nombre, email))
        db.commit()
    
    # Obtener todos los usuarios para mostrar
    db = get_db()
    cursor = db.cursor()
    cursor.execute('SELECT * FROM usuarios')
    usuarios = cursor.fetchall()
    
    return render_template('index.html', usuarios=usuarios)

if __name__ == '__main__':
    init_db()  # Asegurarse de que la tabla existe
    app.run(debug=True)
```
